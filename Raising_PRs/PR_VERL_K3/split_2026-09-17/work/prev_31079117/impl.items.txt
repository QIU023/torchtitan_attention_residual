 1483 + def _cu_seqlens_from_positions(positions):
 1484 +     """Packed offsets ``[0, ..., T]`` of a folded ``[T]`` (or ``[1, T]``) stream from its

 1540 + def _lora_prefix(mod_name: str, sd: dict) -> str:
 1541 +     """The state-dict prefix of a LoRA wrapper reached at ``mod_name``.

 1545 +     adapters ``<prefix>.lora_a.weight`` (a Linear), the earlier wrapper
 1546 +     ``<prefix>.lora_a``; both are accepted, and an unknown wrapper raises rather

 1553 +             return candidate

 1556 +         f"{stripped!r}); an unrecognised module wrapper is in the path"
